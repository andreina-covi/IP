    \begin{frame}[fragile,t]{Variables}
        \begin{itemize}
            \item Una variable es un espacio de memoria que almacena un valor bajo un nombre.
            \item Se crean mediante asignación directa. Ejemplo: \mintinline{python}{nombre = valor}.
            \item No pueden comenzar con un número.
            \item No permiten espacios (se recomienda \mintinline{python}{_} como separador).
            \item Se recomienda el formato \mintinline{python}{snake_case} (ej: \mintinline{python}{nombre_completo}).
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{Variables}
        \begin{itemize}
            \item Los nombres de las variables son sensibles a mayúsculas y minúsculas (ej: \mintinline{python}{nombre} $\neq$ \mintinline{python}{Nombre}).
            \item No pueden ser palabras reservadas del lenguaje.
            \item Deben ser descriptivos del valor que almacenan.
        \end{itemize}
        Ejemplos:
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            cantidad_estudiantes = 25          # Tipo entero
            nota = 3.5                         # Tipo flotante
            nombre_completo = "Maria López"    # Tipo texto
            aprobado = True                    # Tipo booleano
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{print()}}
        \begin{itemize}
            \item Muestra información en pantalla (salida de datos).
            \item Sintaxis básica: \mintinline{python}{print(valor)}.
            \item Puede mostrar texto, números, variables o combinaciones de estos.
            \item Permite concatenar con \mintinline{python}{,} (agrega un espacio automáticamente) o 
            con \mintinline{python}{+} (solo entre strings).
        \end{itemize}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            print("Hola mundo!")
        \end{minted}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            nombre = "Juan"
            edad = 20
            print("Hola", nombre)   # Hola Juan
            print("Tu edad es", edad) # Tu edad es 20
            print("Hola", nombre, "tu edad es", edad) # Hola Juan tu edad es 20
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{input()}}
        \begin{itemize}
            \item Se usa para recibir información del usuario (entrada de datos).
            \item Sintaxis básica: \mintinline{python}{variable =input(mensaje)}.
            \item Siempre devuelve un string \mintinline{python}{str} aunque el usuario escriba un número.
            \item Para convertir a otro tipo de dato se tiene que hacer el casting (conversión de tipos).
            \item Ejemplo: \mintinline{python}{edad = int(input("Ingrese su edad: "))}.
        \end{itemize}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            nombre = input("Ingrese su nombre: ")
            print("Hola", nombre) # Hola Juan
            edad = int(input("Ingrese su edad: "))
            print("Tu edad es", edad) # Tu edad es 20
        \end{minted}
    \end{frame}

            INICIO

    \begin{frame}{Resumen}
        \begin{itemize}
            \item Python es un lenguaje interpretado, de alto nivel y de propósito general.
            \item Python es de tipado dinámico, es decir que infiere automáticamente el tipo de dato de la variable.
            \item La indentación es parte de la sintaxis de Python, no solo estética.
            \item Una variable es un espacio en memoria que almacena un valor y tiene un nombre.
            \item Los nombres de las variables deben ser descriptivos y no contener espacios.
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile, t]{Resumen}
        \begin{itemize}
            \item Los tipos de datos más comunes son: \mintinline{python}{int, float, string, bool}.
            \item Se puede verificar el tipo con la función \mintinline{python}{type()}.
            \item Los operadores aritméticos son: \mintinline{python}{+, -, *, /, **, //, \%}.
            \item Los comparadores son: \mintinline{python}{<, >, <=, >=, ==, !=} y devuelven un valor booleano.
        \end{itemize}
    \end{frame}
