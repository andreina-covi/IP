            with open("reporte.txt", "w", encoding="utf-8") as archivo:
                archivo.write("Reporte de notas\n")
                archivo.write("Estudiante: Josh\n")
        \end{minted}

        \begin{itemize}
            \item \mintinline{python}{write()} recibe un \mintinline{python}{str} como argumento.
            \item No agrega un salto de línea automáticamente (\mintinline{python}{\n}).
            \item Los valores de cualquier tipo diferente a string se tienen que convertir a \mintinline{python}{str}.
        \end{itemize}

        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            with open("reporte.txt", "w", encoding="utf-8") as archivo:
                promedio = 36.8
                archivo.write(promedio)               # TypeError
        \end{minted}
    \end{frame}

    \begin{frame}[fragile, t]{Procesamiento de contenido}
        Las líneas leídas incluyen caracteres que pueden no ser relevantes, como el salto de línea (\mintinline{python}{\n}, espacios o tabulaciones).

        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            linea = " 6.8\n "
            dato = linea.strip()
            print(dato)   # 6.8
        \end{minted}

        \begin{itemize}
            \item \mintinline{python}{strip()} elimina los espacios en blanco (incluidos \mintinline{python}{\n, \t}) al inicio yb al final de la cadena.
            \item \mintinline{python}{strip()} no modifica la cadena original si no que devuelve una nueva cadena.
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile, t]{Separación y conversión de tipos}
        Cuando una línea contiene varios campos, \mintinline{python}{split()} los separa según un delimitador.

        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            linea = "Hugo,17,4.5\t"
            campos = linea.strip().split(",")   # ['6.8', '6.1', '4.5']

            nombre = campos[0]
            edad = int(campos[1])
            nota = float(campos[2])
            print(nombre, edad, nota)   # Hugo 17 4.5
        \end{minted}
    \end{frame}

    \begin{frame}[fragile, t]{Ejercicio: Reporte de notas}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            filename = "notas.csv"
            lista_nombres = [], lista_notas = [], promedios = []
            with open(filename, "r", encoding="utf-8") as a_ent:
                a_ent.readline()           # descarta el encabezado
                for linea in a_ent:
                    campos = linea.strip().split(",")
                    nombre = campos[1]
                    nota = float(campos[2])
                    if nombre in lista_nombres:
                        index = lista_nombres.index(nombre)
                        lista_notas[index].append(float(nota))
                    else:
                        lista_nombres.append(nombre)
                        lista_notas.append([float(nota)]))
        \end{minted}
    \end{frame}

    \begin{frame}[fragile, t]{Ejercicio: Reporte de notas}

            with open("reporte.txt", "w", encoding="utf-8") as a_sal:
                a_sal.write("nombre,nota\n") # crea el encabezado
                for i in range(len(lista_nombres)):
                    nombre = lista_nombres[i]
                    notas = lista_notas[i]
                    promedio = sum(notas) / len(notas)
                    promedios.append(promedio)
                    a_sal.write(nombre + "," + str(promedio) + "\n")
                promedio_general = sum(promedios) / len(promedios)
                a_sal.write("promedio_general:" + str(promedio_general))
        \end{minted}

    \begin{frame}[fragile, t]{Buenas prácticas}
        \begin{itemize}
            \item Usar \mintinline{python}{with} para garantizar el cierre del archivo.
            \item Especificar el modo y la codificación de forma explícita, aunque coincidan con los valores por defecto.

            \item Leer el archivo una sola vez y almacenar lo necesario en variables o listas.
            \item Escribir en un archivo distinto al de entrada, para no perder los datos originales si el programa falla.

        \end{itemize}
