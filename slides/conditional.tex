    \begin{frame}{Condición}
        \begin{itemize}
            \item Una condición permite al programa \textbf{tomar decisiones}
            \item Una condición evalua una expresión como \mintinline{python}{True o False}
            \item Para formar condiciones se usan comparadores como: \mintinline{python}{>, <, ==, !=}
            \item Las condiciones suelen trabajar con variables, números o textos
            \item El resultado de la condición define qué instrucciones se ejecutan
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{Condición}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}

            nota >= 4                   # aprobado o reprobado
            temperatura < 36.0          # tiene fiebre o no
            numero_soldados != 100
        \end{minted}
    \end{frame}

    \begin{frame}{Control de flujo}
        \begin{itemize}
            \item Permite que, según el resultado, el programa continue por un camino u otro
            \item El programa no siempre ejecuta lo mismo
            \item Puede cambiar su camino según la condición
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{if - elif - else}}
        \begin{itemize}
            \item Permite tomar decisiones entre varias opciones
            \item Las condiciones se evaluan una por una y en orden
            \item \mintinline{python}{if} inicia la primera condición
            \item \mintinline{python}{elif} agrega condiciones adicionales
            \item \mintinline{python}{else} representa el caso final cuando ninguna se cumple
        \end{itemize}
        El orden importa. \textbf{Cuando una condición se cumple, las demas ya no se revisan.}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{if - elif - else} - Ejemplos}
        1. Clasificación de puntaje
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            puntaje = 6
            if puntaje >= 6.5:
                print("Rendimiento excelente")
            elif puntaje >= 5:
                print("Rendimiento aceptable")
            else:
                print("Debe reforzar su entrenamiento")
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{if - elif - else} - Ejemplos}
        2. Evaluación de asistencia a instrucción
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            asistencias = 5
            if asistencias == 5:
                print("Asistencia completa")
            elif asistencias >= 3:
                print("Asistencia parcial")
            else:
                print("Asistencia insuficiente")
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{Funciones útiles: \mintinline{python}{abs(), round(), max(), min()}}
        \begin{itemize}
            \item Son funciones integradas de Python, es decir, ya vienen disponibles
            \item Reciben uno o más valores y devuelven un resultado
            \item Simplifican la escritura del código
            \item Ayudan a calcular, ajustar o comparar valores
            \item Pueden utilizarse solas o dentro de expresiones, condiciones y asignaciones
        \end{itemize}

    \begin{frame}[fragile,t]{\mintinline{python}{abs()}}
        \begin{itemize}
            \item Devuelve el valor absoluto de un número
            \item Si el número es negativo, lo convierte en positivo
            \item Si el número ya es positivo, lo deja igual
            \item Útil cuando importa la magnitud del valor, no si es positivo o negativo
            \item Se usa con frecuencia para calcular diferencias, errores o distancias
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{abs()} - Ejemplos}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            temperatura = -5

        \end{minted}

        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            diferencia = abs(8 - 13)
            print(diferencia)         # Imprime 5 como salida
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{round()}}
        \begin{itemize}
            \item Redondea un número
            \item Puede redondear al entero más cercano
            \item También puede redondear a una cantidad específica de decimales
            \item Útil cuando un resultado tiene muchos decimales y queremos hacerlo más claro
            \item Se usa con frecuencia en promedios, medidas, tiempos, distancias y porcentajes
            \item Ayuda a presentar resultados de forma más ordenada
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{max()}}
        \begin{itemize}
            \item Devuelve el valos más grande
            \item Puede comparar dos o más valores
            \item También puede usarse con una colección de datos
            \item Es útil cuando necesitamos identificar el mayor resultado, la mayor distancia, el mayor puntaje o el valor más alto
            \item Ayuda a simplificar el código
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{max()} - Ejemplos}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            max(valor1, valor2, valor3)
        \end{minted}
        
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}

        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}

    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{min()}}
        \begin{itemize}
            \item Devuelve el valor más pequeño
            \item Puede comparar dos o más valores
            \item También puede usarse con una colección de datos
            \item Es útil cuando necesitamos identificar la menor distancia, el menor puntaje, el tiempo más corto o el valor más bajo
            \item Evita escribir muchas comparaciones manuales
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{\mintinline{python}{min()} - Ejemplos}

    \begin{frame}[fragile,t]{Resumen}
        \begin{itemize}
            \item Las condiciones permiten evaluar una situación
            \item Una condición siempre se evalua como \mintinline{python}{True} o \mintinline{python}{False}
            \item El control de flujo usa el resultado de la condición para decidir qué instrucciones ejecutar
            \item \mintinline{python}{if} ejecuta un bloque si la condición es verdadera
            \item \mintinline{python}{else} permite un camino alternativo
            \item \mintinline{python}{elif} permite evaluar varias posibilidades
            \item La indentación es fundamental
            \item Funciones como \mintinline{python}{abs(), round(), max() y min()} qyudan a procesar datos antes de decidir
        \end{itemize}
