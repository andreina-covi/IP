    \begin{frame}{Repaso breve}
        \begin{itemize}
            \item \mintinline{python}{if} ejecuta un bloque si la condición es verdadera
            \item \mintinline{python}{else} se usa cuando no cumple la condición
            \item \mintinline{python}{elif} permite evaluar otra condición
        \end{itemize}
    \end{frame}

    \begin{frame}{\mintinline{python}{if-elif-else}}
        \begin{itemize}
            \item Permite evaluar varias condiciones en orden
            \item El programa entra solo en el primer bloque verdadero
            \item Si ninguna condición se cumple, se ejecuta \mintinline{python}{else}
            \item El orden de las condiciones importa
        \end{itemize}
    \end{frame}

    \begin{frame}{Ejemplo}
        \begin{itemize}
            \item si nota $>=$ 6.5 -> Excelente
            \item si nota $>=$ 5.5 -> Muy bien
            \item si nota $>=$ 4 -> Aprobado
            \item si no -> Reforzamiento
        \end{itemize}
    \end{frame}

    \begin{frame}{\mintinline{python}{ifs} anidados}
        \begin{itemize}
            \item Un \mintinline{python}{if} anidado es un \mintinline{python}{if} dentro de otro \mintinline{python}{if}
            \item Se usa cuando una decisión depende de una condición previa
            \item La segunda evaluación ocurre solo si la primera lo permite
            \item Representa decisiones jerárquicas
            \item Ayuda a modelar procesos por pasos
            \item Debe cuidarse mucho la indentación
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{Operadores lógicos}
        A veces no queremos una decisión dentro de otra, sino evaluar varias condiciones al mismo tiempo en una sola expresión; para eso se usan los operadores lógicos: \mintinline{python}{and, or, not}

        Los operadores lógicos permiten construir condiciones más complejas a partir de condiciones simples.
    \end{frame}

    \begin{frame}{Operadores lógicos: \mintinline{python}{and, or, not}}
        \mintinline{python}{and}
        \begin{itemize}
            \item Ambas condiciones deben ser verdaderas
            \item Se usa cuando se deben cumplir varios requisitos al mismo tiempo
        \end{itemize}
        \mintinline{python}{or}
        \begin{itemize}
            \item Basta con que una condición sea verdadera
            \item Se usa cuando se existen caminos alternativos
        \end{itemize}
        \mintinline{python}{not}
        \begin{itemize}
            \item Invierte el valor lógico
            \item Convierte verdadero en falso, y el falso en verdadero
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{Operador \mintinline{python}{and}}
        La expresión es verdadera si \textbf{ambas condiciones} son verdaderas
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            condicion1 and condicion2
        \end{minted}
        Se usa cuando el problema exige que se cumplan \textbf{varios requisitos al mismo tiempo}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            edad = 20
            tiene_permiso = True

            if edad >= 18 and tiene_permiso:
                print("Puede conducir")
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{Operador \mintinline{python}{or}}
        La expresión es verdadera si \textbf{al menos una} condición es verdaderas
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            condicion1 or condicion2
        \end{minted}
        Se usa cuando existen \textbf{caminos alternativos} para que algo ocurra
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            es_oficial = False
            tiene_permiso_especial = True
            
            if es_oficial or tiene_permiso_especial:
                print("Puede ingresar")
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{Operador \mintinline{python}{not}}
        Sirve para \textbf{invertir} el valor lógico de una condición
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            not condicion
        \end{minted}
        Ejemplo:
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            registrado = False

            if not registrado:
                print("No puede ingresar al sistema")
        \end{minted}
    \end{frame}

    \begin{frame}{Tablas de verdad}
        La tabla de verdad ayuda a analizar, de forma ordenada, todas las combinaciones posibles entre condiciones y el resultado final de la expresión
        \begin{itemize}
            \item Organiza los valores \textbf{Verdadero (V)} y \textbf{Falso (F)} de una o más condiciones
            \item Muestra el resultado de aplicar operadores lógicos como \mintinline{python}{and, or, not}
            \item Ayuda a comprender cómo funcionan las condiciones compuestas
            \item Permite analizar todas las posibilidades sin omitir casos
        \end{itemize}
    \end{frame}
