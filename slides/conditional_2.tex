    \begin{frame}[fragile,t]{Contenido}
        \begin{enumerate}
            \item Sintaxis básica de \mintinline{python}{if-elif-else}
            \item Condicionales con operadores lógicos
            \item \mintinline{python}{if} anidados: múltiples condicionales
        \end{enumerate}
    \end{frame}

    \begin{frame}{Repaso breve}
        \begin{itemize}
            \item \mintinline{python}{if} ejecuta un bloque si la condición es verdadera
            \item \mintinline{python}{elif} permite evaluar otra condición
            \item \mintinline{python}{else} se usa cuando no cumple la condición
            \item El orden de las condiciones importa
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{Estructura general de \mintinline{python}{if-elif-else}}
        \begin{itemize}
            \item Estructura de \mintinline{python}{if}:
            \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
                if condicion:
                    print("Se cumple la condición")
            \end{minted}
            \item Estructura de \mintinline{python}{if-elif}:
            \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}   
                if condicion1:
                    print("Se cumple la condición")
                elif condicion2:
                    print("Se cumple la segunda condición")
            \end{minted}
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{Estructura general de \mintinline{python}{if-elif-else}}
        \begin{itemize}
            \item Estructura de \mintinline{python}{if-elif-else}:
            \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            if condicion1:
                print("Se cumple la primera condición")
            elif condicion2:
                print("Se cumple la segunda condición")
            elif condicionN:
                print("Se cumple la N-ésima condición")
            else:
                print("No se cumple ninguna condición")
            \end{minted}
        \end{itemize}
    \end{frame}

    \begin{frame}{Actividad}
        \begin{enumerate}
            \item Pedir un número e indicar si es positivo, negativo o cero
            \item Pedir dos números e indicar si el primero es divisible por el segundo
            \item Pedir dos números y mostrar cuál es el mayor
            \item Pedir un número de tres cifras e indicar si es capicúa o no
        \end{enumerate}
    \end{frame}

    \begin{frame}[fragile,t]{Condicional con operador \mintinline{python}{and}}
        \begin{itemize}
            \item Con \mintinline{python}{and} se pueden evaluar múltiples condiciones
            \item Todas las condiciones deben ser verdaderas para que se ejecute el bloque
            \item Si alguna condición es falsa, el bloque no se ejecuta
        \end{itemize}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            if condicion1 and condicion2:
                print("Se cumplen ambas condiciones")
        \end{minted}
        Ejemplo:
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            profundidad = 35   # metros
            tiempo = 25        # minutos
            
            if profundidad > 30 and tiempo > 20:
                print("Riesgo alto: requiere parada de descompresion")
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{Condicional con operador \mintinline{python}{and} - Ejemplo}
        En Chile la red eléctrica domiciliaria opera a \textbf{220\,V / 50\,Hz}. 
        Los equipos suelen tolerar un margen de aprox. $\pm$5--10\,V; ambas condiciones
        (mínimo y máximo) deben cumplirse a la vez.
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            voltaje = 225   # volts
 
            if voltaje >= 210 and voltaje <= 230:
                print("Voltaje dentro de rango seguro")
            else:
                print("Voltaje fuera de rango: revisar instalacion") 
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{Condicional con operador \mintinline{python}{or}}
        \begin{itemize}
            \item Con \mintinline{python}{or} se pueden evaluar múltiples condiciones
            \item Solo se necesita que una condición sea verdadera para que se ejecute el bloque
            \item Si todas las condiciones son falsas, el bloque no se ejecuta
        \end{itemize}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            if condicion1 or condicion2:
                print("Se cumple alguna condición")
        \end{minted}
        Ejemplo:
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            edad = 17
            tiene_licencia = True
            if edad >= 18 or tiene_licencia:
                print("La persona puede conducir")
        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{Condicional con operador \mintinline{python}{or} - Ejemplo}
        Chile se ubica en el Cinturón de Fuego del Pacífico y
        concentra parte importante de la actividad sísmica
        mundial. Los sistemas de alerta temprana consideran
        \emph{magnitud} \textbf{o} \emph{profundidad} del sismo.

        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}

            if magnitud >= 7.5 or profundidad_km <= 30:

        \end{minted}
    \end{frame}

    \begin{frame}[fragile,t]{Condicional con operador \mintinline{python}{not}}
        Con \mintinline{python}{not} se puede negar una condición

        Ejemplo:

    \begin{frame}{\mintinline{python}{ifs} anidados}
        \begin{itemize}
            \item Un \mintinline{python}{if} anidado es un \mintinline{python}{if} dentro de otro \mintinline{python}{if}
            \item Se usa cuando una decisión depende de una condición previa
            \item La segunda evaluación ocurre solo si la primera lo permite
            \item Representa decisiones jerárquicas
            \item Ayuda a modelar procesos por pasos
            \item Debe cuidarse mucho la indentación
        \end{itemize}
    \end{frame}

    \begin{frame}[fragile,t]{Estructura general}
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            if condicion1:
                if condicion2:
                    print("Se cumplen ambas condiciones")
        \end{minted}
        Con \mintinline{python}{else} puede verse así:
        \begin{minted}[xleftmargin=-2cm, frame=leftline, fontsize=\small]{python}
            if condicion1:
                if condicion2:
                    print("Puede continuar")
                else:
                    print("Falla la segunda condicion")
            else:
                print("Falla la primera condicion")
        \end{minted}
    \end{frame}

    \begin{frame}{Alerta por calidad de aire - Ejercicio}
        Dado el nivel de PM2.5 en el aire ($\mu$g/m³) y si la persona consultante pertenece 
        a un grupo de riesgo (niños, adultos mayores o personas con enfermedades respiratorias), 
        determina el mensaje de alerta que debe mostrar la aplicación.

        \begin{itemize}
            \item Si el PM2.5 supera 55 $\mu$g/m³ (nivel considerado dañino según estándares de 
            calidad del aire), y la persona es de grupo de riesgo → alerta máxima.
            \item Si supera ese nivel pero la persona no es de grupo de riesgo $\rightarrow$ alerta moderada.
            \item Si no supera ese nivel $\rightarrow$ aire aceptable.
        \end{itemize}
    \end{frame}

    \begin{frame}{Resumen}
        \begin{itemize}
            \item \mintinline{python}{if} ejecuta un bloque solo cuando una
                  condición es \mintinline{python}{True}.
            \item \mintinline{python}{and} $\rightarrow$ \textbf{todas} las
                  condiciones deben cumplirse.
            \item \mintinline{python}{or} $\rightarrow$ \textbf{al menos una}
                  condición debe cumplirse.
            \item \mintinline{python}{not} $\rightarrow$ \textbf{invierte} el
                  valor de verdad.
            \item Estos operadores modelan decisiones reales: alertas
                  médicas, sísmicas, estructurales y de comportamiento
                  animal.
        \end{itemize}
    \end{frame}
